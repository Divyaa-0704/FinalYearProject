    Contact_Info VARCHAR(255),

CREATE TABLE nationalparks (
    park_id INT IDENTITY(1,1) PRIMARY KEY,

);

    employee_id INT IDENTITY(1,1) PRIMARY KEY,

CREATE TABLE animals (
    Animal_ID INT IDENTITY(1,1) PRIMARY KEY,
    Species VARCHAR(50) NOT NULL,
    Name VARCHAR(50),
    Age INT,
    Status VARCHAR(20) CHECK (Status IN ('Present', 'Rescued', 'Deceased')) NOT NULL,
    Health_Condition VARCHAR(20) CHECK (Health_Condition IN ('Good', 'Critical', 'Recovering')) NOT NULL,
    Park_ID INT,
    FOREIGN KEY (Park_ID) REFERENCES NationalParks(park_id)
);

-- Creating the Suppliers Table
CREATE TABLE suppliers (
    supplier_id INT IDENTITY(1,1) PRIMARY KEY,    -- supplier_id as an auto-increment column
    name VARCHAR(100) NOT NULL,
    contact_email VARCHAR(100) UNIQUE NOT NULL,    -- contact_email is unique
    contact_phone VARCHAR(15),
    address TEXT
);

-- Creating the Procurement Table
CREATE TABLE procurement (
    procurement_id INT IDENTITY(1,1) PRIMARY KEY,  -- procurement_id as an auto-increment column
    park_id INT,
    supplier_id INT,
    item_name VARCHAR(100) NOT NULL,
    quantity INT,
    cost DECIMAL(10, 2),
    procurement_date DATE,
    FOREIGN KEY (park_id) REFERENCES NationalParks(park_id),
    FOREIGN KEY (supplier_id) REFERENCES Suppliers(supplier_id)
);
